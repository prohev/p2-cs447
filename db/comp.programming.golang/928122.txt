c.cntFail &gt;= c.cfg.MaxFail { // Closed =&gt; Open c.stateOpen() } if c.state == stHalfOpen &amp;&amp; c.cntFail &gt; 0 { // HalfOpen =&gt; Open c.stateOpen() } if c.state == stHalfOpen &amp;&amp; c.cntSucc &gt;= c.cfg.MinSucc { // HalfOpen =&gt;
